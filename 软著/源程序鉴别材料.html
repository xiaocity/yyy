  const fresh=SHEEP.filter(s=>!save.dex[s.id]);
  let pick;
  if(fresh.length&&(save.pity||0)>=PITY_N){
    /* 保底：优先补抽中档位里缺的，那一档满了就补最低稀有度里缺的 ——
       保底不该成为白嫖传说的通道 */
    const inTier=fresh.filter(s=>s.r===want);
    const lowest=Math.min.apply(null,fresh.map(s=>s.r));
    pick=rnd(inTier.length?inTier:fresh.filter(s=>s.r===lowest));
  }else{
    /* 平时也向「缺的」倾斜：稀有度先掷，档内七成概率优先给没有的那只。

    const pool=SHEEP.filter(s=>s.r===want), fp=pool.filter(s=>!save.dex[s.id]);
    pick=(fp.length&&Math.random()<0.7)?rnd(fp):rnd(pool);
  }
  if(!pick) return null;
  if(save.dex[pick.id]){ save.dex[pick.id].s++; save.pity=(save.pity||0)+1; return {sheep:pick,d
up:true}; }
  save.dex[pick.id]={s:0}; save.pity=0;
  return {sheep:pick,dup:false};
}
